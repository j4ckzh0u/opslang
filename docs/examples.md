# OpsLang 示例教程

本目录包含 OpsLang 的实用示例脚本。每个示例都包含完整代码、运行命令和预期输出。

## 运行方式

所有示例通过 `opsctl run` 命令执行：

```bash
opsctl run examples/示例文件.ops
```

## 1. 系统信息采集

用真实 SDK 调用采集系统信息，超过阈值自动告警，输出结构化报告。

**文件**: `examples/collect_system_info.ops`

```ops
// 系统信息采集脚本
// 使用真实 SDK 调用采集本机系统信息

let hostname_info = sys.hostname()
let cpu_usage = sys.cpu.usage()
let cpu_count = sys.cpu.count()
let mem_info = sys.memory.info()
let disk_info = sys.disk.usage("/")
let load_info = sys.load()
let os_info = sys.os()
let uptime_info = sys.uptime()
let net_ifaces = sys.net.interfaces()

// 条件告警
if cpu_usage.percent > 80 {
    alert("CPU 使用率过高: " + str(cpu_usage.percent) + "%")
}

if mem_info.used_percent > 90 {
    alert("内存使用率过高: " + str(mem_info.used_percent) + "%")
}

if disk_info.used_percent > 90 {
    alert("磁盘使用率过高: " + str(disk_info.used_percent) + "%")
}

// 输出网络接口名称（C 风格下标循环）
let iface_names = []
for let i = 0; i < len(net_ifaces); i = i + 1 {
    let iface = net_ifaces[i]
    iface_names = iface_names + [iface.name]
}

// 计算运行时长（小时）
let uptime_hours = uptime_info.uptime / 3600.0

// 结构化报告
report {
    host: hostname_info.hostname,
    os: os_info.os,
    platform: os_info.platform,
    kernel: os_info.kernel_version,
    cpu_percent: cpu_usage.percent,
    cpu_cores_logical: cpu_count.logical,
    cpu_cores_physical: cpu_count.physical,
    mem_total: mem_info.total,
    mem_used: mem_info.used,
    mem_percent: mem_info.used_percent,
    disk_percent: disk_info.used_percent,
    load_1m: load_info.load1,
    uptime_hours: uptime_hours,
    network_interfaces: iface_names
}

log("系统信息采集完成")
```

**运行命令**:

```bash
opsctl run examples/collect_system_info.ops
```

**预期输出**（数值随本机实际情况变化）:

```
ALERT: CPU 使用率过高: 91.2%        # 仅当 CPU > 80% 时出现
{
  "host": "server-01",
  "os": "linux",
  "platform": "ubuntu",
  "kernel": "6.8.0-40-generic",
  "cpu_percent": 12.5,
  "cpu_cores_logical": 8,
  "cpu_cores_physical": 4,
  ...
}
系统信息采集完成
```

> 说明：数据来自真实 SDK 调用（`sys.cpu.usage()` 500ms 采样窗口），不同机器、不同时刻输出不同。

## 2. 文件批量操作

批量创建目录、写入文件、列出目录、计算校验和、清理。

**文件**: `examples/batch_file_ops.ops`

```ops
// 批量文件操作示例
// 演示批量创建目录、写入文件、列出目录、计算校验和、清理

print("=== 批量文件操作 ===")

// 基础目录
let base_dir = "/tmp/opslang_batch_test"

// 第一步：创建目录结构
let subdirs = ["logs", "data", "config", "reports"]
let created_dirs = []

for let i = 0; i < len(subdirs); i = i + 1 {
    let dir_path = base_dir + "/" + subdirs[i]
    file.mkdir(dir_path)
    created_dirs = created_dirs + [dir_path]
    print("创建目录: " + dir_path)
}

// 第二步：在每个子目录中写入文件
for let i = 0; i < len(subdirs); i = i + 1 {
    let file_path = base_dir + "/" + subdirs[i] + "/output.txt"
    let content = "Directory: " + subdirs[i] + "\nGenerated by OpsLang batch_file_ops\n"
    let result = file.write(file_path, content)
    print("写入文件: " + file_path + " (" + str(result.size) + " bytes)")
}

// 第三步：列出基础目录内容
let listing = file.list(base_dir)
print("---")
print("目录列表 (" + base_dir + "):")
for let i = 0; i < len(listing.files); i = i + 1 {
    let entry = listing.files[i]
    print("  " + entry.name + " (dir: " + str(entry.is_dir) + ", size: " + str(entry.size) + ")")
}

// 第四步：计算所有文件的校验和
print("---")
print("文件校验和:")
for let i = 0; i < len(subdirs); i = i + 1 {
    let file_path = base_dir + "/" + subdirs[i] + "/output.txt"
    let cs = file.checksum(file_path, "sha256")
    print("  " + file_path + " -> " + cs.checksum)
}

// 第七步：清理
for let i = 0; i < len(subdirs); i = i + 1 {
    file.delete(base_dir + "/" + subdirs[i] + "/output.txt")
}

report {
    total_dirs: len(subdirs),
    dirs_created: created_dirs,
    operations: ["mkdir", "write", "list", "checksum", "delete"],
    status: "completed"
}
```

**运行命令**:

```bash
opsctl run examples/batch_file_ops.ops
```

**预期输出**（校验和随内容变化）:

```
=== 批量文件操作 ===
创建目录: /tmp/opslang_batch_test/logs
创建目录: /tmp/opslang_batch_test/data
...
写入文件: /tmp/opslang_batch_test/logs/output.txt (58 bytes)
...
文件校验和:
  /tmp/opslang_batch_test/logs/output.txt -> 3f2a...e1
...
```

## 3. 服务健康检查

用真实 SDK 检查内存、磁盘、进程等健康状态，按阈值分级告警。

**文件**: `examples/service_health_check.ops`

```ops
// 服务健康检查脚本
// 使用 SDK 检查系统资源、磁盘、进程等健康状态
// 兼容 macOS 和 Linux（不使用 service.* 函数，因为 macOS 无 systemd）

print("=== 服务健康检查 ===")

let host = sys.hostname()
let mem = sys.memory.info()
let disk = sys.disk.usage("/")
let os_info = sys.os()
let ts = time.now()

// 检查内存健康状态
let mem_status = "healthy"
if mem.used_percent > 95 {
    mem_status = "critical"
    alert("[CRITICAL] 内存使用率过高: " + str(mem.used_percent) + "%")
} else if mem.used_percent > 80 {
    mem_status = "warning"
    alert("[WARNING] 内存使用率偏高: " + str(mem.used_percent) + "%")
}

print("主机: " + host.hostname)
print("内存使用: " + str(mem.used_percent) + "% [" + mem_status + "]")
print("内存总量: " + str(mem.total / 1024 / 1024) + " MB")

// 检查磁盘健康
let disk_status = "healthy"
if disk.used_percent > 95 {
    disk_status = "critical"
} else if disk.used_percent > 80 {
    disk_status = "warning"
}
print("磁盘 /: " + str(disk.used_percent) + "% 已用 [" + disk_status + "]")

// 检查运行时间（使用 sys.os() 的 uptime 字段）
let uptime_hours = os_info.uptime / 3600
print("系统运行时长: " + str(uptime_hours) + " 小时")

print("---")
print("系统进程:")

print(ps_result.stdout)

// 汇总报告
report {
    host: host.hostname,
    timestamp: ts.iso8601,
    memory_percent: mem.used_percent,
    memory_status: mem_status,
    disk_percent: disk.used_percent,
    disk_status: disk_status,
    uptime_hours: uptime_hours,
    os: os_info.os,
    platform: os_info.platform,
    kernel: os_info.kernel_version,
    overall_status: "checked"
}
```

## 4. 应用部署流程

用真实文件操作模拟多步骤应用部署：建目录、写配置、校验完整性、幂等标记、清理。

**文件**: `examples/deploy_app.ops`（节选）

```ops
// deploy_app.ops - Simulate application deployment using file operations
// 所有操作使用 /tmp/，兼容 macOS 和 Linux

print("=== Application Deployment Simulation ===")

let hostname = sys.hostname()
let now = time.now()
let app_name = "myapp"
let app_version = "2.1.0"
let deploy_dir = "/tmp/ops-example-deploy"
let release_dir = deploy_dir + "/releases/" + app_version
let config_dir = deploy_dir + "/config"

// --- Step 1: Create directory structure ---
file.mkdir(deploy_dir)
file.mkdir(deploy_dir + "/releases")
file.mkdir(release_dir)
file.mkdir(config_dir)

// --- Step 2: Write configuration files ---
let app_config = "{\"name\": \"" + app_name + "\", \"version\": \"" + app_version + "\", \"workers\": 4, \"port\": 8080}"
let config_path = config_dir + "/app.json"
file.write(config_path, app_config)

// --- Step 3: Deploy application files ---
let app_binary = "#!/bin/sh\necho \"" + app_name + " v" + app_version + " running\"\n"
let binary_path = release_dir + "/" + app_name
file.write(binary_path, app_binary)

// --- Step 4: Verify file integrity with checksums ---
let cs_binary = file.checksum(binary_path, "sha256")
let cs_config = file.checksum(config_path, "sha256")

let all_exist = true
let check_files = [binary_path, config_path]
for let i = 0; i < len(check_files); i = i + 1 {
    let ex = file.exists(check_files[i])
    if !ex.exists {
        all_exist = false
        alert("Missing file: " + check_files[i])
    }
}

// --- Step 5: Write deployment marker（ensure 幂等：重复运行不再变更） ---
let marker_path = deploy_dir + "/.deployed"
file.write(marker_path, "{\"version\": \"" + app_version + "\"}")

ensure file.exists(marker_path).exists {
    print("  SHOULD NOT SEE THIS - marker already exists")
}

// --- Step 6: Final summary ---
report {
    host: hostname.hostname,
    app: app_name,
    version: app_version,
    deployed_at: now.iso8601,
    files_verified: all_exist,
    binary_checksum: cs_binary.checksum,
    config_checksum: cs_config.checksum,
    status: "deployed"
}

log("Deployment completed: " + app_name + " v" + app_version)

// Cleanup（完整脚本末尾会清理全部临时文件）
```

**运行命令**:

```bash
opsctl run examples/deploy_app.ops
```

**预期输出**（校验和随内容变化；完整脚本共 6 步并在结束时清理临时文件）:

```
=== Application Deployment Simulation ===
Host: server-01
App: myapp v2.1.0
...
[4/6] Verifying integrity...
  All files verified!
  Marker verified (idempotent)
...
Deployment completed: myapp v2.1.0 on server-01
Cleanup complete - all temporary files removed
```

## 6. ensure 幂等操作

声明式幂等：check → apply → verify → notify。条件为假时执行动作体，再次验证仍为假则报错；可选的 `notify` 表达式只在实际发生变更后求值。

**文件**: `examples/ensure_example.ops`（节选）

```ops
// ensure 模式实现 check -> apply -> verify 语义:
//   1. CHECK: 评估条件
//   2. APPLY: 条件为假时执行动作体
//   3. VERIFY: 重新评估条件，仍为假则报错
//   4. NOTIFY: 仅在实际发生变更后求值（可选）

let base_dir = "/tmp/ops-example-ensure"
let config_file = base_dir + "/config.json"

// --- 第一次运行：资源不存在，ensure 创建一切 ---
print("--- First run: creating resources ---")

ensure file.exists(base_dir).exists {
    print("  Creating base directory: " + base_dir)
    file.mkdir(base_dir)
}

ensure file.exists(config_file).exists {
    print("  Creating config file: " + config_file)
    file.write(config_file, "{\"app\": \"demo\", \"version\": \"1.0\"}")
}

// --- 第二次运行：一切就绪，ensure 动作体不会执行 ---
print("--- Second run: idempotent (no changes expected) ---")

let changes = 0

ensure file.exists(config_file).exists {
    print("  SHOULD NOT SEE THIS - config already exists")
    changes = changes + 1
}

if changes == 0 {
    print("  No changes made - idempotency verified!")
} else {
    alert("Unexpected changes detected: " + str(changes))
}

// 校验和
let cs = file.checksum(config_file, "sha256")

report {
    resources_ensured: 4,
    changes_on_rerun: changes,
    idempotent: changes == 0,
    config_checksum: cs.checksum
}
```

**运行命令**:

```bash
opsctl run examples/ensure_example.ops

# 干运行：apply 只打印不执行
opsctl run --dry-run examples/ensure_example.ops
```

**预期输出**（首次运行）:

```
=== Ensure (Idempotent) Pattern Demo ===
--- First run: creating resources ---
  Creating base directory: /tmp/ops-example-ensure
  Creating config file: /tmp/ops-example-ensure/config.json
...
--- Second run: idempotent (no changes expected) ---
  No changes made - idempotency verified!
...
Ensure pattern demo completed
```

> 说明：第二次运行时所有条件都已满足，动作体（含 `SHOULD NOT SEE THIS`）不会执行，`changes` 保持 0。

## 7. 闭包与函数

演示闭包捕获外部变量、嵌套函数与真实 SDK 数据结合。

**文件**: `examples/functions_closures.ops`（节选）

```ops
// functions_closures.ops - Closures and nested functions with SDK data

// Closure: make a threshold checker for CPU usage.
// 内部函数捕获外部的 threshold
fn make_cpu_checker(threshold) {
    fn check() {
        let cpu = sys.cpu.usage()
        if cpu.percent > threshold {
            return "ABOVE " + str(threshold) + "% (actual: " + str(cpu.percent) + "%)"
        }
        return "BELOW " + str(threshold) + "% (actual: " + str(cpu.percent) + "%)"
    }
    return check
}

let strict_checker = make_cpu_checker(30)
let relaxed_checker = make_cpu_checker(80)

print("Strict check: " + strict_checker())
print("Relaxed check: " + relaxed_checker())

// 对一组路径求磁盘用量之和（C 风格下标循环）
fn disk_usage_sum(paths) {
    let total = 0
    for let i = 0; i < len(paths); i = i + 1 {
        let d = sys.disk.usage(paths[i])
        total = total + d.used
    }
    return total
}

let sum = disk_usage_sum(["/", "/tmp"])
print("Total disk used (/ + /tmp): " + str(sum / 1024 / 1024) + " MB")
```


```
Strict check: BELOW 30% (actual: 12.5%)
Relaxed check: BELOW 80% (actual: 12.6%)
Total disk used (/ + /tmp): 123456 MB
```

### 真正的远程执行（opsctl deploy）

`task "名字" on <目标>` 的目标路由在 `opsctl deploy` 下生效；`opsctl run` 遇到带 `on` 的 task 会报错提示改用 deploy：

```bash
# 脚本中的 task "collect" on "web*" 只路由到匹配 web* 的目标
opsctl deploy my_task.ops --targets root@web1,root@web2,root@db1 --key ~/.ssh/id_rsa
```

runner 模式（线性任务体）按 `on` 选择器分发；含控制流/ensure/parallel 的脚本自动走 AOT 模式（全语言，但 task 级 `on` 路由不支持，会报错）。详见 `docs/cli-reference.md`。

## 9. 幂等收敛（ensure 家族，对标 Ansible）

### 9.1 ensure_idempotency_proof.ops —— 幂等性自证

在真实文件系统上跑两轮完全相同的收敛，第二轮必须零变更，且**脚本内带断言**（不满足会真的 alert）：

```bash
opsctl run examples/ensure_idempotency_proof.ops
```

```text
--- 第 1 轮（首次收敛，预期 changed=true）---
file.ensure(directory): changed=true actions=[mkdir]
file.ensure(touch):     changed=true actions=[create]
--- 第 2 轮（重复收敛，预期 changed=false 且零动作）---
file.ensure(directory): changed=false msg=directory already up to date
file.ensure(touch):     changed=false
--- 权限漂移恢复 ---
漂移后收敛: changed=true mode=0640
```

### 9.2 ensure_user_group.ops —— 用户/组生命周期（真机验证）

完整的运维生命周期：建组 → 建用户 → 校验 → 制造 shell 漂移 → ensure 自动收敛 → 清理。变更需要 Linux + root（脚本内置 `/run` 写探测），其他平台显式 SKIP。

```bash
opsctl run examples/ensure_user_group.ops            # macOS/非 root：诚实 SKIP
opsctl deploy examples/ensure_user_group.ops --inventory hosts.yaml --mode aot --auto-approve
```

真机（Ubuntu 22.04，AOT 模式）真实输出：

```text
[group.ensure] changed=true msg=group created gid=1002
[group.ensure 复跑] changed=false msg=group already exists
[user.ensure] changed=true msg=user created shell=/usr/sbin/nologin
[漂移演练] 改为 bash 后 ensure 收敛: changed=true shell=/usr/sbin/nologin
[user.absent] changed=true msg=user removed
```

### 9.3 ensure_service.ops —— 服务状态收敛（真机验证）

不硬编码假设装了哪个服务：探测候选单元（cron/sshd/ssh/systemd-timesyncd），把第一个真实存在的收敛为 started + enabled。真机验证过两个方向：

```text
# cron 已在运行（幂等路径）：
[service.ensure(started)]  changed=false actions=[] msg=service "cron" already active

# 手工 systemctl stop cron 之后（真实收敛路径）：
[before] active=false enabled=true
[service.ensure(started)]  changed=true actions=[start] msg=service "cron" converged to started
[after]  active=true enabled=true
```

### 9.4 ensure_app_layout.ops —— 应用目录基线收敛

任何平台可跑（/tmp 下真实操作）：目录骨架 + 配置文件 + 权限基线 + lineinfile，复跑零变更断言。

### 9.5 remote_ensure_fleet.ops / remote_fleet_teardown.ops —— 舰队供给与下线

deploy 专用剧本（对标 Ansible playbook + inventory）：

```bash
# 供给：所有主机报 facts；group=root 的主机收敛服务账号 + cron
opsctl deploy examples/remote_ensure_fleet.ops --inventory hosts.yaml --auto-approve
# 幂等实证：立即重复部署，所有 changed=false

# 下线（与供给完全对称）
opsctl deploy examples/remote_fleet_teardown.ops --inventory hosts.yaml --auto-approve
```

inventory 按组路由（对标 Ansible `hosts:` 字段）：

```yaml
hosts:
  - name: host2
    host: 10.0.0.12
    user: root
    group: root      # task "..." on "root" 路由到这台
```

### 真实执行政策

所有示例遵守同一铁律：**要么真实执行，要么显式 SKIP 并给出真实原因**。变更操作内置前提探测（平台/root/systemd 单元），探测本身也是真实系统调用（如向 `/run` 写探针文件验证 root 能力）。不存在打印假成功的路径；每个示例都被 `cmd/opsctl/examples_e2e_test.go` 在 CI 中真实执行。
